快速开始

git clone https://github.com/LiHongwei-cn/lihongwei-cn.git

# 进入 global-specs 目录
cd lihongwei-cn/global-specs

# 运行安装脚本
# macOS/Linux:
bash install.sh
# Windows:
install.bat
